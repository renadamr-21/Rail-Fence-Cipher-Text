.data
msg1:        .asciiz " Welcome to rail fence cipher program \n"
startmsg:    .asciiz " Do you want to perform encryption/decryption using Rail Fence Cipher? ( yes=1 , 0=no ): "
keymsg:      .asciiz " Enter the numeric key (number of rails): "
errormsg:    .asciiz " Invalid key. Try again and enter key >= 2 \n"
textmsg:     .asciiz " Enter the plaintext message: "
decmsg:      .asciiz " Do you want to decrypt the ciphertext? ( yes=1 , 0=no ): "
out_plain:   .asciiz " Plaintext: "
out_cipher:  .asciiz " Generated Ciphertext: "
out_decrypted:.asciiz " Decrypted Plaintext: "
newline:     .asciiz "\n"
exitmsg:     .asciiz " Exiting program... \n"

plaintext:   .space 256
ciphertext:  .space 256
decrypted:   .space 256

.text
.globl main

main:
    li  $v0, 4
    la  $a0, msg1
    syscall

main_loop:
    li  $v0, 4
    la  $a0, startmsg
    syscall

    li  $v0, 5
    syscall
    beq $v0, $zero, exit_program

get_key:
    li  $v0, 4
    la  $a0, keymsg
    syscall

    li  $v0, 5
    syscall
    move $s1, $v0          

    blt  $s1, 2, key_error
    j    get_text

key_error:
    li  $v0, 4
    la  $a0, errormsg
    syscall
    j    get_key


get_text:
    li  $v0, 4
    la  $a0, textmsg
    syscall

    li  $v0, 8
    la  $a0, plaintext
    li  $a1, 256
    syscall

    la  $t0, plaintext
    li  $s0, 0            

len_loop:
    lb  $t1, 0($t0)
    beq $t1, 10, len_done
    beq $t1, 0, len_done
    addi $s0, $s0, 1
    addi $t0, $t0, 1
    j   len_loop

len_done:
    sb  $zero, 0($t0)
    jal encrypt
    li  $v0, 4
    la  $a0, out_plain
    syscall

    li  $v0, 4
    la  $a0, plaintext
    syscall

    li  $v0, 4
    la  $a0, newline
    syscall

    li  $v0, 4
    la  $a0, out_cipher
    syscall
    la  $a0, ciphertext
    syscall
    la  $a0, newline
    syscall

    li  $v0, 4
    la  $a0, decmsg
    syscall

    li  $v0, 5
    syscall
    beq $v0, $zero, main_loop

    jal decrypt

    li  $v0, 4
    la  $a0, out_decrypted
    syscall
    la  $a0, decrypted
    syscall
    la  $a0, newline
    syscall

    j   main_loop

exit_program:
    li  $v0, 4
    la  $a0, exitmsg
    syscall

    li  $v0, 10
    syscall

encrypt:
    la  $t0, plaintext
    la  $t1, ciphertext

    sub $t2, $s1, 1
    sll $t2, $t2, 1         

    li  $t3, 0              

enc_rail_loop:
    beq $t3, $s1, enc_done
    li  $t4, 0              

enc_char_loop:
    bge $t4, $s0, enc_next_rail

    div $t4, $t2
    mfhi $t5                

    sub $t6, $s1, 1
    blt $t5, $s1, enc_down
    sub $t5, $t5, $t6
    sub $t5, $t6, $t5

enc_down:
    bne $t5, $t3, enc_skip

    la  $t7, plaintext
    add $t7, $t7, $t4
    lb  $t8, 0($t7)
    sb  $t8, 0($t1)
    addi $t1, $t1, 1

enc_skip:
    addi $t4, $t4, 1
    j   enc_char_loop

enc_next_rail:
    addi $t3, $t3, 1
    j   enc_rail_loop

enc_done:
    sb  $zero, 0($t1)
    jr  $ra


decrypt:
    la  $t0, ciphertext
    la  $t1, decrypted

    sub $t2, $s1, 1
    sll $t2, $t2, 1       
    li  $t3, 0              

dec_rail_loop:
    beq $t3, $s1, dec_done
    li  $t4, 0              

dec_char_loop:
    bge $t4, $s0, dec_next_rail

    div $t4, $t2
    mfhi $t5

    sub $t6, $s1, 1
    blt $t5, $s1, dec_down
    sub $t5, $t5, $t6
    sub $t5, $t6, $t5

dec_down:
    bne $t5, $t3, dec_skip

    lb  $t8, 0($t0)
    la  $t7, decrypted
    add $t7, $t7, $t4
    sb  $t8, 0($t7)
    addi $t0, $t0, 1

dec_skip:
    addi $t4, $t4, 1
    j   dec_char_loop

dec_next_rail:
    addi $t3, $t3, 1
    j   dec_rail_loop

dec_done:
    la  $t7, decrypted
    add $t7, $t7, $s0
    sb  $zero, 0($t7)
    jr  $ra
